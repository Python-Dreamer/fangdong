function bldgOptions(selectedId){var s=selectedId||'';return(cache.buildings||[]).map(function(b){return'<option value="'+b.id+'"'+(b.id===s?' selected':'')+'>'+esc(b.name)+'</option>'}).join('')}
function quickAddBuilding(){if(!canEdit()){toast(i18n.t('common.readOnly'),'w');return}var name=prompt(i18n.t('building.quickAddPh'),'');if(!name)return;name=name.trim();if(!name)return;if((cache.buildings||[]).some(function(b){return b.name.toLowerCase()===name.toLowerCase()})){toast(i18n.t('building.nameExists'),'e');return}sb.from('buildings').insert({owner_id:getOwnerId(),name:name}).then(function(r){if(r.error){toast(r.error.message,'e');return}toast(i18n.t('building.saveSuccess'));return loadAllData()}).then(function(){if(!cache)return;var sel=document.getElementById('fRB');if(sel){var nb=(cache.buildings||[]).find(function(b){return b.name.toLowerCase()===name.toLowerCase()});if(nb){sel.innerHTML='<option value="">'+i18n.t('building.unassigned')+'</option>'+bldgOptions(nb.id);sel.value=nb.id}}})}
function showBuildingManager(){if(!canEdit()){toast(i18n.t('common.readOnly'),'w');return}function listHtml(){var blds=cache.buildings||[];if(!blds.length)return'<div class="empty" style="padding:24px"><div class="ei">🏢</div><h3>'+i18n.t('building.add')+'</h3><p style="font-size:12px;color:var(--ts)">'+i18n.t('building.quickAddPh')+'</p></div>';return'<div style="display:flex;flex-direction:column;gap:8px">'+blds.map(function(b){var cnt=cache.rooms.filter(function(r){return r.building_id===b.id}).length;return'<div style="display:flex;align-items:center;justify-content:space-between;padding:10px 12px;border:1px solid var(--bd);border-radius:8px;background:var(--card,#fff)"><div><div style="font-weight:600;font-size:14px">'+esc(b.name)+'</div><div style="font-size:11px;color:var(--ts);margin-top:2px">'+cnt+' '+i18n.t('building.rooms')+(b.address?' · '+esc(b.address):'')+'</div></div><div style="display:flex;gap:6px"><button class="btn btn-sm btn-gh" onclick="showBuildingEdit(\''+b.id+'\')">'+i18n.t('common.edit')+'</button><button class="btn btn-sm btn-gh" style="color:var(--d)" onclick="delBuilding(\''+b.id+'\')">'+i18n.t('common.delete2')+'</button></div></div>'}).join('')+'</div>'}openModal('<div class="m-hd"><h3>'+i18n.t('building.manage')+'</h3><button class="m-x" onclick="closeModal()">×</button></div><div class="m-bd"><button class="btn btn-p" style="width:100%;margin-bottom:12px" onclick="showBuildingEdit(null)">+ '+i18n.t('building.add')+'</button><div id="bldgList">'+listHtml()+'</div></div>')}
function showBuildingEdit(id){var b=id?(cache.buildings||[]).find(function(x){return x.id===id}):null;openModal('<div class="m-hd"><h3>'+(b?i18n.t('building.edit'):i18n.t('building.add'))+'</h3><button class="m-x" onclick="closeModal()">×</button></div><div class="m-bd"><div class="fg"><label>'+i18n.t('building.name')+'<span class="req">*</span></label><input class="fc" id="fBName" value="'+esc(b?b.name:'')+'" placeholder="'+i18n.t('building.namePh')+'"></div><div class="fg"><label>'+i18n.t('building.addr')+'</label><input class="fc" id="fBAddr" value="'+esc(b?(b.address||''):'')+'" placeholder="'+i18n.t('building.addrPh')+'"></div><div class="fg"><label>'+i18n.t('building.notes')+'</label><textarea class="fc" id="fBNotes" placeholder="'+i18n.t('building.notesPh')+'">'+esc(b?(b.notes||''):'')+'</textarea></div><div style="border-top:1px dashed var(--bd);margin:12px 0;padding-top:12px"><div style="font-size:13px;font-weight:600;margin-bottom:4px;color:var(--p)">💧 '+i18n.t('meter.title')+'</div><p style="font-size:11px;color:var(--ts);margin:0 0 10px">'+i18n.t('building.priceHint')+'</p><div class="fr"><div class="fg"><label>'+i18n.t('building.coldPrice')+'</label><input class="fc" id="fBCold" type="number" step="0.01" value="'+(b&&b.cold_water_price!=null?b.cold_water_price:'')+'"></div><div class="fg"><label>'+i18n.t('building.hotPrice')+'</label><input class="fc" id="fBHot" type="number" step="0.01" value="'+(b&&b.hot_water_price!=null?b.hot_water_price:'')+'"></div></div><div class="fg"><label>'+i18n.t('building.elecPrice')+'</label><input class="fc" id="fBElec" type="number" step="0.01" value="'+(b&&b.electricity_price!=null?b.electricity_price:'')+'"></div></div></div><div class="m-ft"><button class="btn btn-gh" onclick="showBuildingManager()">'+i18n.t('common.cancel')+'</button><button class="btn btn-p" id="saveBuildingBtn" onclick="doSaveBuilding(\''+(b?b.id:'')+'\')">'+i18n.t('common.save')+'</button></div>')}
async function doSaveBuilding(id){var name=document.getElementById('fBName').value.trim();if(!name){toast(i18n.t('building.nameRequired'),'e');return}if((cache.buildings||[]).some(function(x){return x.id!==id&&x.name.toLowerCase()===name.toLowerCase()})){toast(i18n.t('building.nameExists'),'e');return}function nv(elId){var v=document.getElementById(elId).value;if(v==='')return null;var n=parseFloat(v);return isNaN(n)?null:n}var obj={name:name,address:document.getElementById('fBAddr').value.trim()||null,notes:document.getElementById('fBNotes').value.trim()||null,cold_water_price:nv('fBCold'),hot_water_price:nv('fBHot'),electricity_price:nv('fBElec')};var btn=document.getElementById('saveBuildingBtn');if(btn){btn.textContent=i18n.t('common.saving');btn.disabled=true}try{var r;if(id){r=await sb.from('buildings').update(obj).eq('id',id)}else{obj.owner_id=getOwnerId();r=await sb.from('buildings').insert(obj)}if(r.error){toast(r.error.message,'e');return}toast(i18n.t('building.saveSuccess'));await loadAllData();showBuildingManager()}finally{if(btn){btn.textContent=i18n.t('common.save');btn.disabled=false}}}
function delBuilding(id){var b=(cache.buildings||[]).find(function(x){return x.id===id});if(!b)return;var cnt=cache.rooms.filter(function(r){return r.building_id===id}).length;if(cnt>0){toast(i18n.t('building.deleteHasRooms'),'w');return}confirmDialog(i18n.t('building.deleteConfirm'),i18n.t('common.noRecover'),async function(){var r=await sb.from('buildings').delete().eq('id',id);if(r.error){toast(r.error.message,'e');return}toast(i18n.t('common.deleteSuccess'));await loadAllData();showBuildingManager()})}
